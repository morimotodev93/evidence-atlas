function permitted(raw) {
  const u = new URL(raw);
  return u.origin === origin && (u.pathname === '/demo' || /^\/demo\/research\/[^/]+$/.test(u.pathname) || u.pathname.startsWith('/_next/static/') || u.pathname === '/favicon.ico');
}
function addCount(obj, key) { obj[key] = (obj[key] || 0) + 1; }
let browser;
async function context(viewport) {
  const ctx = await browser.newContext({ viewport, deviceScaleFactor: 1, colorScheme: 'light', serviceWorkers: 'block' });
  await ctx.route('**/*', async route => {
    const req = route.request(), u = new URL(req.url());
    if (req.method() !== 'GET' || !permitted(req.url())) {
      addCount(network.blocked, req.method() + ' ' + u.origin + u.pathname);
      return route.abort('blockedbyclient');
    }
    network.allowed++; addCount(network.methods, req.method()); addCount(network.paths, u.pathname);
    if (!network.allowedOrigins.includes(u.origin)) network.allowedOrigins.push(u.origin);
    try {
      const response = await route.fetch({ maxRedirects: 0, timeout: 30000 });
      const location = response.headers().location;
      if (location && !permitted(new URL(location, req.url()).href)) { network.redirectBlocked++; return route.abort('blockedbyclient'); }
      addCount(network.responses, response.status());
      await route.fulfill({ response });
    } catch { network.errors++; await route.abort(); }
  });
  await ctx.routeWebSocket('**/*', ws => { network.websocketBlocked++; ws.close(); });
  return ctx;
}
async function environment(page) {
  return page.evaluate(() => ({ userAgent: navigator.userAgent, width: innerWidth, height: innerHeight, dpr: devicePixelRatio, scale: visualViewport.scale, rootZoom: getComputedStyle(document.documentElement).zoom, bodyZoom: getComputedStyle(document.body).zoom, rootClass: document.documentElement.className, light: matchMedia('(prefers-color-scheme: light)').matches, bodyBackground: getComputedStyle(document.body).backgroundColor, font: getComputedStyle(document.body).fontFamily, fonts: document.fonts.status }));
}
async function load(page, href, title) {
  const response = await page.goto(origin + href, { waitUntil: 'networkidle', timeout: 45000 });
  await page.evaluate(() => document.fonts.ready);
  if (response?.status() !== 200 || page.url() !== origin + href || (await page.locator('h1').textContent()) !== title || await page.getByText('Read-only demo', { exact: true }).count() !== 1) throw new Error('STOP: HTTP/content preflight mismatch');
  await page.evaluate(install);
  await page.mouse.move((await page.evaluate(() => innerWidth)) - 1, 1);
  return response;
}
async function order(page) {
  return page.locator('a').evaluateAll(es => es.filter(e => e.getClientRects().length && getComputedStyle(e).visibility !== 'hidden').map(e => ({ selector: window.__pr22.selector(e), text: e.textContent.trim(), href: e.getAttribute('href') })));
}
async function focused(page, expected, direction) {
  // Wait only; never focus/scroll the target. Require four stable animation frames
  // for scroll position/rect and stop after two seconds if animation persists.
  const stable = await page.evaluate(async () => {
    let previous, equal = 0;
    for (let i = 0; i < 120; i++) {
      await new Promise(requestAnimationFrame);
      const e = document.activeElement, r = e?.getBoundingClientRect();
      const value = JSON.stringify([scrollX, scrollY, r?.top, r?.bottom]);
      equal = value === previous ? equal + 1 : 0; previous = value;
      if (equal >= 4) return true;
    }
    return false;
  });
  const f = await page.evaluate(() => {
    const v = window.__pr22.focus(); delete v.samples;
    const e = document.activeElement, s = getComputedStyle(e);
    return { ...v, boundingRect: (() => { const r=e.getBoundingClientRect(); return { left:r.left,right:r.right,top:r.top,bottom:r.bottom,width:r.width,height:r.height }; })(), scrollMarginTop: s.scrollMarginTop, scrollMarginBottom: s.scrollMarginBottom, bottomGap: Math.min(...v.rects.map(r => innerHeight-r.bottom)), topGap: Math.min(...v.rects.map(r => r.top-(v.kind==='Header'?0:v.header.bottom))) };
  });
  f.direction = direction; f.expectedSelector = expected.selector; f.orderPass = f.target === expected.selector; f.layoutStable = stable;
  f.result = stable && f.orderPass && f.focusVisible && f.indicator && f.exposed && f.rects.length > 0 && f.clearanceIssues.length === 0 ? 'Pass' : 'Fail';
  return f;
}
async function press(page, direction, expected) { await page.keyboard.press(direction); return focused(page, expected, direction); }
